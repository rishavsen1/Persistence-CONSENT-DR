\newcommand{\stitle}{\bfseries}%

\begin{tikzpicture}[
  font=\footnotesize,
  every node/.style={transform shape},
  stage/.style={rectangle, rounded corners=3pt, draw=black!55, thick,
                fill=white, align=left, inner sep=6pt,

  flow/.style={-latex, thick, black!70},

% ===================== STAGES =====================
\node[stage] (pledge) {%
{\stitle 1. Commit the daily pledge}\\[2pt]
A day ahead, the building forecasts how much surviving buffer it will
have and commits to a peak cap $\FSL_d$ it can defend, earning a capacity
payment.\\[3pt]
{\seqn $\psihat \!\to\! \buffhat{d} \!\to\! \Omeps$ \eqref{eq:pledgeset};\
set $\FSL_d(\lambda)$ \eqref{eq:pledgedepth}; earn $\Cinc{d}$
\eqref{eq:dr}}};

\node[stage, right=0.6cm of pledge] (arrive) {%
{\stitle 2. Negotiate with each driver}\\[2pt]
On arrival, a driver states their needs. The system offers compensated
flexibility options consistent with the pledge, and a free top-up if
there is room.\\[3pt]
{\seqn observe $\tarr,\Earr$, report $\Ereq,\treq$; price by $\Uflex$
\eqref{eq:uflex} $\Rightarrow \Eneg,\tdep$; offer $\Rightarrow
\accept{v}{k}$}};

\node[stage, right=0.6cm of arrive] (control) {%
{\stitle 3. Charge in real time}\\[2pt]
Every few minutes the controller sets charging rates at lowest cost,
topping up batteries only into spare capacity so banking never creates a
new peak.\\[3pt]
{\seqn minimize \eqref{eq:chobj} s.t.\ SoC/floor/cap
\eqref{eq:socbounds}--\eqref{eq:fslslack}; bank only if $\Ptot{t} <
\min(\Pmaxhat,\text{cap})$}};

\node[stage, below=1.55cm of control] (event) {%
{\stitle 4. Deliver during an event}\\[2pt]
If the grid calls an event, the controller discharges the banked energy
to hold the load under the pledged cap; any overshoot is penalized.\\[3pt]
{\seqn discharge to $\Emin(v)$, keep $\Edep\!\ge\!\Eneg$ \eqref{eq:floor};
slack $\vio{t}$ billed via $\Cpen{d}$ \eqref{eq:dr}}};

\node[stage, left=0.6cm of event] (depart) {%
{\stitle 5. Settle at departure}\\[2pt]
The driver leaves with at least the SoC they agreed to. Energy banked
above that floor is recorded, and the driver is paid for any event-day
help.\\[3pt]
{\seqn guarantee \eqref{eq:floor}; bank $\banked{v}{k}$ \eqref{eq:banked};
settle $\ucost{k}{v}$ \eqref{eq:usercost}, $\Ubuf$ by replay
\eqref{eq:ubuf}}};

\node[stage, left=0.6cm of depart] (night) {%
{\stitle 6. Carry over to next visit}\\[2pt]
Between visits the car is driven, consuming some of the top-up. What
survives becomes tomorrow's deliverable buffer and sharpens the next
forecast.\\[3pt]
{\seqn $\Earr\!=\!\Edep\!-\!\Eext$ \eqref{eq:carryover}; survives
$\survived{v}{k+1}$ \eqref{eq:survived}; form $\buff{d}$ \eqref{eq:buffer},
update $\ohat,\rhohat,\buffhat{d+1}$}};

% ===================== ARROWS =====================
\draw[flow] (pledge) -- (arrive);
